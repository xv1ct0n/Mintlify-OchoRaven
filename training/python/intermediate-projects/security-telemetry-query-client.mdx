# Project 2: Security Telemetry Query Client

**Intent:** Run a bounded hunt query against an authorized Elasticsearch or OpenSearch lab and export an event summary.

**Result:** A working utility, synthetic fixture, expected output, and a verification check. The project is designed for authorized lab data and can be adapted after you inspect the real data/API contract.

## Learning objectives

1. Bound the query — Build an exact term query with an explicit result size and ascending timestamp sort. Use a keyword field when the mapping needs exact matching. Limited hits do not constitute a full export.
2. Separate transport — The HTTP function owns HTTPS, credentials, CA trust, timeout, and request encoding. Extraction and summarization work on plain dictionaries, so they can be checked offline.
3. Keep provenance — Preserve hit IDs, the generated query, event types, and first and last timestamps. Put the password in an environment variable and use a trusted internal CA file when needed.
4. Validate offline first — A saved search response is a repeatable fixture. Inspect the report before querying a live lab and compare the field mapping to the backend.

## Step 1: Build the program

Save this complete reference implementation as `telemetry_query.py`. The standard-library projects need no third-party package. The advanced validator and API service declare their dependencies below.

```python
"""Query a lab Elasticsearch/OpenSearch endpoint or an offline response fixture."""
import argparse
import base64
import json
import os
import ssl
import urllib.error
import urllib.parse
import urllib.request
from collections import Counter
from pathlib import Path


def build_query(field, value, limit):
    return {"size": limit, "sort": [{"@timestamp": "asc"}],
            "query": {"term": {field: value}}}


def run_query(endpoint, index, query, username, password, timeout, ca_file):
    if not endpoint.startswith("https://"):
        raise ValueError("endpoint must use HTTPS")
    url = endpoint.rstrip("/") + "/" + urllib.parse.quote(index, safe="-_") + "/_search"
    auth = base64.b64encode(f"{username}:{password}".encode()).decode()
    request = urllib.request.Request(url, data=json.dumps(query).encode(),
        headers={"Content-Type": "application/json", "Authorization": "Basic " + auth}, method="POST")
    try:
        with urllib.request.urlopen(request, timeout=timeout,
                                    context=ssl.create_default_context(cafile=ca_file)) as response:
            return json.load(response)
    except urllib.error.HTTPError as exc:
        raise RuntimeError(f"Search returned HTTP {exc.code}") from exc
    except (urllib.error.URLError, TimeoutError) as exc:
        raise RuntimeError("Search endpoint unavailable or TLS validation failed") from exc


def extract_events(response):
    hits = response.get("hits", {}).get("hits", [])
    if not isinstance(hits, list):
        raise ValueError("Invalid search response: hits.hits must be a list")
    return [{"id": hit.get("_id"), **hit.get("_source", {})} for hit in hits]


def summarize_events(events):
    return {"count": len(events), "by_event_type": dict(Counter(
        str(event.get("event_type", "unknown")) for event in events)),
        "first_timestamp": min((e.get("@timestamp", "") for e in events), default=""),
        "last_timestamp": max((e.get("@timestamp", "") for e in events), default="")}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--fixture", type=Path, help="saved _search JSON response")
    parser.add_argument("--endpoint", default=os.getenv("SEARCH_URL"))
    parser.add_argument("--index", default="lab-events")
    parser.add_argument("--field", default="event_type.keyword")
    parser.add_argument("--value", default="authentication_failure")
    parser.add_argument("--limit", type=int, default=100)
    parser.add_argument("--timeout", type=float, default=10)
    parser.add_argument("--ca-file", help="CA bundle for internal TLS certificates")
    parser.add_argument("--output", type=Path, default=Path("query-results.json"))
    args = parser.parse_args()
    if not 1 <= args.limit <= 1000:
        parser.error("--limit must be between 1 and 1000")
    query = build_query(args.field, args.value, args.limit)
    if args.fixture:
        response = json.loads(args.fixture.read_text())
    else:
        user, password = os.getenv("SEARCH_USER"), os.getenv("SEARCH_PASSWORD")
        if not args.endpoint or not user or not password:
            parser.error("set SEARCH_URL, SEARCH_USER, SEARCH_PASSWORD or use --fixture")
        response = run_query(args.endpoint, args.index, query, user, password,
                             args.timeout, args.ca_file)
    events = extract_events(response)
    output = {"query": query, "summary": summarize_events(events), "events": events}
    args.output.write_text(json.dumps(output, indent=2) + "\n")
    print(json.dumps(output["summary"], indent=2))


if __name__ == "__main__":
    main()
```

## Step 3: Run and inspect

```bash
python telemetry_query.py --fixture response.json --output query-results.json
```

**Expected:** The summary reports two failures and the output retains the query and both event IDs.

## Step 5: Test the failure path

Try a missing field. For live 401 check access, for certificate failure supply --ca-file, and for zero hits inspect index, mapping, time coverage, and permissions.
