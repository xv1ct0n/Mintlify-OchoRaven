# Project 3: Telemetry Pipeline Health Monitor

**Intent:** Measure freshness, volume, required-field completeness, and duplicate IDs with explicit health thresholds.

**Result:** A working utility, synthetic fixture, expected output, and a verification check. The project is designed for authorized lab data and can be adapted after you inspect the real data/API contract.

## Learning objectives

1. Explicit baseline — Configuration lists expected sources, warning and critical age, a volume window, minimum count, and required fields. Thresholds are local assumptions.
2. UTC time — Require timezone-aware timestamps and normalize to UTC. The --now option makes the lab reproducible; live runs use current UTC.
3. Source checks — Group events by source, find newest valid event, count recent records, check required fields and duplicate IDs, and report absent sources.
4. Severity and operations — Critical covers absent or stale telemetry and zero expected volume; warning covers delay, partial low volume, missing fields, or duplicates. Exit codes 0, 1, and 2 support scheduled monitoring.

## Step 1: Build the program

Save this complete reference implementation as `telemetry_monitor.py`. The standard-library projects need no third-party package. The advanced validator and API service declare their dependencies below.

```python
"""Assess source freshness, volume, field completeness, and duplicate IDs."""
import argparse
import json
from collections import Counter, defaultdict
from datetime import datetime, timezone
from pathlib import Path


def parse_time(value):
    dt = datetime.fromisoformat(value.replace("Z", "+00:00"))
    if dt.tzinfo is None:
        raise ValueError("timestamp must include a timezone")
    return dt.astimezone(timezone.utc)


def assess(events, config, now):
    groups = defaultdict(list)
    for event in events:
        groups[event.get("source")].append(event)
    report = []
    for name, spec in config["sources"].items():
        rows, reasons = groups[name], []
        valid = []
        for event in rows:
            try:
                valid.append(parse_time(event["timestamp"]))
            except (KeyError, TypeError, ValueError):
                reasons.append("unparseable timestamp")
        ages = [(now - stamp).total_seconds() / 60 for stamp in valid]
        age = min(ages, default=None)
        window = spec["window_minutes"]
        volume = sum(0 <= x <= window for x in ages)
        if age is None or age > spec["critical_after_minutes"]:
            status = "critical"; reasons.append("stale or absent events")
        elif age > spec["warning_after_minutes"]:
            status = "warning"; reasons.append("delayed events")
        else:
            status = "healthy"
        if volume < spec["minimum_volume"]:
            status = "critical" if volume == 0 else max(status, "warning", key=["healthy", "warning", "critical"].index)
            reasons.append("volume below baseline")
        missing = sum(any(field not in event for field in spec["required_fields"]) for event in rows)
        if missing:
            if status == "healthy": status = "warning"
            reasons.append(f"{missing} records missing required fields")
        ids = [row.get("event_id") for row in rows if row.get("event_id")]
        duplicates = sum(n - 1 for n in Counter(ids).values() if n > 1)
        if duplicates:
            if status == "healthy": status = "warning"
            reasons.append(f"{duplicates} duplicate event IDs")
        report.append({"source": name, "status": status, "last_age_minutes": age,
                       "volume": volume, "missing_fields": missing,
                       "duplicate_ids": duplicates, "reasons": sorted(set(reasons))})
    return report


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("events", type=Path, help="JSONL events with source, timestamp, event_id")
    parser.add_argument("config", type=Path, help="JSON baseline and thresholds")
    parser.add_argument("--now", help="ISO timestamp for reproducible lab runs")
    parser.add_argument("--output", type=Path, default=Path("health.json"))
    args = parser.parse_args()
    events = [json.loads(line) for line in args.events.read_text().splitlines() if line.strip()]
    now = parse_time(args.now) if args.now else datetime.now(timezone.utc)
    report = assess(events, json.loads(args.config.read_text()), now)
    args.output.write_text(json.dumps(report, indent=2) + "\n")
    for row in report:
        print(f"{row['source']}: {row['status']} ({', '.join(row['reasons']) or 'ok'})")
    raise SystemExit(2 if any(r["status"] == "critical" for r in report) else 1 if any(r["status"] == "warning" for r in report) else 0)


if __name__ == "__main__":
    main()
```
